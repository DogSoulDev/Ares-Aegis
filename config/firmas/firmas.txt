# === EXPRESIONES REGULARES ===
# Detectar eval con base64
REGEX:eval\s*\(\s*base64_decode

# Detectar variables superglobales de PHP en contexto sospechoso
REGEX:\$_[A-Z]+\[['\"][^'\"]*['\"]\]

# Detectar system() con variables superglobales
REGEX:system\s*\([^)]*\$_

# Detectar descargas directas de scripts
REGEX:(?:wget|curl)\s+https?://[^\s]+\.(sh|py|pl|exe|bat)

# Detectar shells reversos
REGEX:/bin/(ba)?sh\s+-[il]

# Detectar netcat con parámetros de listener
REGEX:nc\s+.*-[lp]\s+\d+

# Detectar ofuscación común en JavaScript
REGEX:eval\s*\(\s*unescape

# Detectar PowerShell codificado
REGEX:powershell\s+.*-enc\s+[A-Za-z0-9+/=]+
